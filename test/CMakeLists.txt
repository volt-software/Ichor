include(CTest)
include(Catch)

option(ICHOR_SKIP_EXTERNAL_TESTS "When running tests in containers, etcd and redis aren't running. This option skips those." OFF)

file(GLOB_RECURSE PROJECT_TEST_SOURCES ${ICHOR_TOP_DIR}/test/*.cpp)
# Not tests themselves, but compiled into every test: Support/TestEnvironment.cpp gives each test process its own working directory, so that ctest
# can run them in parallel
list(FILTER PROJECT_TEST_SOURCES EXCLUDE REGEX "/test/Support/")
# A source of each test instead of a library: a library linking ichor would put libichor.a on the link line twice, which gcc's LTO with mold
# doesn't survive (undefined symbols)
set(ICHOR_TEST_SUPPORT_SOURCES ${ICHOR_TOP_DIR}/test/Support/TestEnvironment.cpp)
if(CMAKE_CXX_COMPILER_ID STREQUAL "GNU")
    set(CMAKE_CXX_FLAGS "${CMAKE_CXX_FLAGS} -Wno-maybe-uninitialized ") # Catch2 has something that triggers gcc.
endif()

foreach(filename ${PROJECT_TEST_SOURCES})
    get_filename_component(testname ${filename} NAME_WE)

    if((WIN32 OR APPLE) AND (${testname} STREQUAL "TcpTests" OR ${testname} STREQUAL "HttpHostSocketTests"))
        continue()
    endif()


    if(${testname} MATCHES "^OpenSSL" AND NOT ICHOR_USE_OPENSSL)
        continue()
    endif()


    if(NOT ${testname} STREQUAL "HttpEndToEndTests" AND NOT ${testname} STREQUAL "EtcdTests" AND NOT ${testname} STREQUAL "AsyncFileIOTests_uring")
        add_executable(${testname} ${filename} ${ICHOR_TEST_SUPPORT_SOURCES})
        target_link_libraries(${testname} ${CMAKE_THREAD_LIBS_INIT})
        target_link_libraries(${testname} ichor)
        target_link_libraries(${testname} Catch2::Catch2WithMain)
        target_compile_definitions(${testname} PUBLIC CATCH_CONFIG_FAST_COMPILE)
    endif()

    # We want to build them, but not run/discover them.
    if(ICHOR_SKIP_EXTERNAL_TESTS AND (${testname} STREQUAL "EtcdTests" OR ${testname} STREQUAL "RedisTests" OR ${testname} STREQUAL "HttpEndToEndTests"))
        continue()
    endif()
    if(ICHOR_SKIP_EXTERNAL_TESTS)
        target_compile_definitions(${testname} PUBLIC ICHOR_SKIP_EXTERNAL_TESTS)
    endif()
    if(${testname} STREQUAL "AsyncFileIOTests_uring")
        if(ICHOR_USE_LIBURING AND NOT (ICHOR_SKIP_EXTERNAL_TESTS AND ICHOR_AARCH64))
            add_executable(${testname} ${filename} ${ICHOR_TEST_SUPPORT_SOURCES})
            target_link_libraries(${testname} ${CMAKE_THREAD_LIBS_INIT})
            target_link_libraries(${testname} ichor)
            target_link_libraries(${testname} Catch2::Catch2WithMain)
            target_compile_definitions(${testname} PUBLIC CATCH_CONFIG_FAST_COMPILE)
            catch_discover_tests(${testname})
            continue()
        else()
            continue()
        endif()
    endif()
    if(ICHOR_USE_LIBURING AND NOT (ICHOR_SKIP_EXTERNAL_TESTS AND ICHOR_AARCH64))
        if(${testname} STREQUAL "TcpTests" OR ${testname} STREQUAL "HttpHostSocketTests" OR ${testname} STREQUAL "ServicesTests" OR ${testname} STREQUAL "CoroutineTests" OR ${testname} STREQUAL "HttpEndToEndTests" OR ${testname} STREQUAL "EtcdTests")
            add_executable(${testname}_uring ${filename} ${ICHOR_TEST_SUPPORT_SOURCES})
            target_link_libraries(${testname}_uring ${CMAKE_THREAD_LIBS_INIT})
            target_link_libraries(${testname}_uring ichor)
            target_link_libraries(${testname}_uring Catch2::Catch2WithMain)
            target_compile_definitions(${testname}_uring PUBLIC CATCH_CONFIG_FAST_COMPILE TEST_URING)
            if(${testname} STREQUAL "EtcdTests")
                # The tests share the etcd server: its keys, and authentication, which the v2 test turns on and off
                catch_discover_tests(${testname}_uring PROPERTIES RESOURCE_LOCK etcd)
            else()
                catch_discover_tests(${testname}_uring)
            endif()
        endif()
    endif()
    if(ICHOR_USE_SDEVENT AND NOT (ICHOR_SKIP_EXTERNAL_TESTS AND ICHOR_AARCH64))
        if(${testname} STREQUAL "ServicesTests" OR ${testname} STREQUAL "CoroutineTests")
            add_executable(${testname}_sdevent ${filename} ${ICHOR_TEST_SUPPORT_SOURCES})
            target_link_libraries(${testname}_sdevent ${CMAKE_THREAD_LIBS_INIT})
            target_link_libraries(${testname}_sdevent ichor)
            target_link_libraries(${testname}_sdevent Catch2::Catch2WithMain)
            target_compile_definitions(${testname}_sdevent PUBLIC CATCH_CONFIG_FAST_COMPILE TEST_SDEVENT)
            catch_discover_tests(${testname}_sdevent)
        endif()
    endif()
    if(ICHOR_USE_BOOST_BEAST AND NOT (ICHOR_SKIP_EXTERNAL_TESTS AND ICHOR_AARCH64))
        if(${testname} STREQUAL "HttpEndToEndTests" OR ${testname} STREQUAL "EtcdTests")
            add_executable(${testname}_boost ${filename} ${ICHOR_TEST_SUPPORT_SOURCES})
            target_link_libraries(${testname}_boost ${CMAKE_THREAD_LIBS_INIT})
            target_link_libraries(${testname}_boost ichor)
            target_link_libraries(${testname}_boost Catch2::Catch2WithMain)
            target_compile_definitions(${testname}_boost PUBLIC CATCH_CONFIG_FAST_COMPILE TEST_BOOST)
            if(${testname} STREQUAL "EtcdTests")
                catch_discover_tests(${testname}_boost PROPERTIES RESOURCE_LOCK etcd)
            else()
                catch_discover_tests(${testname}_boost)
            endif()
        endif()
    endif()
    if(${testname} STREQUAL "ServicesTests" OR ${testname} STREQUAL "CoroutineTests")
        add_executable(${testname}_ordered ${filename} ${ICHOR_TEST_SUPPORT_SOURCES})
        target_link_libraries(${testname}_ordered ${CMAKE_THREAD_LIBS_INIT})
        target_link_libraries(${testname}_ordered ichor)
        target_link_libraries(${testname}_ordered Catch2::Catch2WithMain)
        target_compile_definitions(${testname}_ordered PUBLIC CATCH_CONFIG_FAST_COMPILE TEST_ORDERED)
        catch_discover_tests(${testname}_ordered)
    endif()

    if(NOT ${testname} STREQUAL "HttpEndToEndTests" AND NOT ${testname} STREQUAL "EtcdTests")
        if(${testname} STREQUAL "RedisTests")
            # The tests share the redis server and its keys
            catch_discover_tests(${testname} PROPERTIES RESOURCE_LOCK redis)
        else()
            catch_discover_tests(${testname})
        endif()
    endif()
endforeach()

# A failing assertion terminates the process, which Catch2 can't test: these check what AssertFailure prints instead.
add_executable(AssertFailure ${ICHOR_TOP_DIR}/test/Support/AssertFailure.cpp)
target_link_libraries(AssertFailure ichor)
set(ICHOR_ASSERT_FAILURE_MESSAGE "Assertion (`value == 1` )?failed: value is 2 instead of 1")
add_test(NAME "AssertTests: ICHOR_ASSERT_ALWAYS prints the condition and message"
        COMMAND ${CMAKE_COMMAND} -DEXE=$<TARGET_FILE:AssertFailure> -DMODE=always "-DEXPECTED=${ICHOR_ASSERT_FAILURE_MESSAGE}" -P ${ICHOR_TOP_DIR}/test/Support/CheckAssertFailure.cmake)
add_test(NAME "AssertTests: ICHOR_ASSERT prints the condition and message when hardening"
        COMMAND ${CMAKE_COMMAND} -DEXE=$<TARGET_FILE:AssertFailure> -DMODE=hardened "-DEXPECTED=${ICHOR_ASSERT_FAILURE_MESSAGE}|isn't checked without hardening" -P ${ICHOR_TOP_DIR}/test/Support/CheckAssertFailure.cmake)
if(ICHOR_HAVE_STD_STACKTRACE AND NOT ICHOR_USE_LIBCPP)
    # " 0# " is the first frame of libstdc++'s formatting of a std::stacktrace
    add_test(NAME "AssertTests: failed assertions print a stack trace"
            COMMAND ${CMAKE_COMMAND} -DEXE=$<TARGET_FILE:AssertFailure> -DMODE=always "-DEXPECTED= 0# " -P ${ICHOR_TOP_DIR}/test/Support/CheckAssertFailure.cmake)
endif()
